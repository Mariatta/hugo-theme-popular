{{- /* The days an event covers. Context = an event page. Returns a dict: start (the `date`), end (the optional `endDate`, the last day, inclusive, when it is later than `date`; otherwise the same day as start), and multi (true for a multi-day event). An `endDate` before `date` is ignored, with a build warning. Mirrors Astro's lib/eventdates.ts. */ -}} {{- $start := .Date -}} {{- $end := $start -}} {{- with .Params.endDate -}} {{- $parsed := try (time.AsTime .) -}} {{- if $parsed.Err }}{{ errorf "%s: endDate %q is not a date; write it like 2027-01-17" $.File.Path (string .) }}{{ end -}} {{- $last := $parsed.Value | default $start -}} {{- if gt ($last.Format "20060102") ($start.Format "20060102") }}{{ $end = $last -}} {{- else if lt ($last.Format "20060102") ($start.Format "20060102") }}{{ warnf "%s: endDate %s is before date %s, so it is ignored" $.File.Path ($last.Format "2006-01-02") ($start.Format "2006-01-02") }}{{ end -}} {{- end -}} {{- return (dict "start" $start "end" $end "multi" (ne ($start.Format "20060102") ($end.Format "20060102"))) -}}
